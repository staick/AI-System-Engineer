class Solution:  
    def maxNumberOfFamilies(self, n: int, reservedSeats: List[List[int]]) -> int:  
        ans = 0  
        group_types = [[1, 1, 2, 1, 0] for _ in range(n)]  
        for s in reservedSeats:  
            i = s[0] - 1  
            if s[1] in (2, 3, 4, 5):  
                group_types[i][0] = 0  
                group_types[i][2] = 0  
            if s[1] in (4, 5, 6, 7):  
                group_types[i][1] = 0  
            if s[1] in (6, 7, 8, 9):  
                group_types[i][2] = 0  
                group_types[i][3] = 0  
        for g in group_types:  
            if g[0]:  
                g[1] = 0  
            if g[1]:  
                g[2] = 0  
                g[3] = 0  
            if g[2]:  
                g[0] = 0  
                g[3] = 0  
            ans += sum(g)  
  
        return ans

class Solution:  
    def maxNumberOfFamilies(self, n: int, reservedSeats: List[List[int]]) -> int:  
        ans = 0  
        group_types = [1, 1, 2, 1, 0]  
        index_set = set()  
        reservedSeats = sorted(reservedSeats, key=lambda x: x[0])  
        index = reservedSeats[0][0]  
        for s in reservedSeats:  
            index_set.add(s[0])  
            if s[0] != index:  
                if group_types[0]:  
                    group_types[1] = 0  
                if group_types[1]:  
                    group_types[2] = 0  
                    group_types[3] = 0  
                if group_types[2]:  
                    group_types[0] = 0  
                    group_types[3] = 0  
                ans += sum(group_types)  
                group_types = [1, 1, 2, 1, 0]  
                index = s[0]  
            if s[1] in (2, 3, 4, 5):  
                group_types[0] = 0  
                group_types[2] = 0  
            if s[1] in (4, 5, 6, 7):  
                group_types[1] = 0  
            if s[1] in (6, 7, 8, 9):  
                group_types[2] = 0  
                group_types[3] = 0  
        if group_types[0]:  
            group_types[1] = 0  
        if group_types[1]:  
            group_types[2] = 0  
            group_types[3] = 0  
        if group_types[2]:  
            group_types[0] = 0  
            group_types[3] = 0  
        ans += sum(group_types)  
        ans += (n - len(index_set)) * 2  
  
        return ans

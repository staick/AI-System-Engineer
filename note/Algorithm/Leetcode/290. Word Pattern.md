class Solution:  
    def wordPattern(self, pattern: str, s: str) -> bool:  
        map_dict = {}  
        s_dict = {}  
        l = s.split(sep=" ")  
        n = len(pattern)  
        if n != len(l):  
            return False  
        for i in range(n):  
            if (pattern[i] in map_dict and map_dict[pattern[i]] != l[i]) or (  
                l[i] in s_dict and s_dict[l[i]] != pattern[i]  
            ):  
                return False  
            map_dict[pattern[i]], s_dict[l[i]] = l[i], pattern[i]  
        return True

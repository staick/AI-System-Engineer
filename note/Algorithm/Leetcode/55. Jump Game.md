class Solution:
	def canJump(self, nums: List[int]) -> bool:
		max_step = 0
		for i,num in enumerate(nums):
			max_step -= 1
			if num ==0 and i == len(nums)-1:
				return True
			if num == 0 and max_step <= 0:
				return False
			max_step = max(max_step, num)
		return True
```

class Solution:
	def canJump(self, nums: List[int]) -> bool:
	mx = 0
	for i, jump in enumerate(nums):
		if i > mx: # 无法到达 i
			return False
		mx = max(mx, i + jump) # 从 i 最右可以跳到 i+jump
	return True
```

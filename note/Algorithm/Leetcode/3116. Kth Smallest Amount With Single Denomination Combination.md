大根堆，超时
```python
class Solution:
    def findKthSmallest(self, coins: List[int], k: int) -> int:
        heap = [-coins[0] * i for i in range(1, k + 1)]
        heapq.heapify(heap)
        for i in coins[1:]:
            for j in range(1, k + 1):
                if -i * j in heap:
                    continue
                elif -i * j > heap[0]:
                    heapq.heappop(heap)
                    heapq.heappush(heap, -i * j)
                else:
                    break
        return -heap[0]
```

使用二分+容斥原理
```python
class Solution:
    def findKthSmallest(self, coins: List[int], k: int) -> int:
	    def check(m):  
		    cnt = 0  
		    n = len(coins)  
		    for i in range(1, 1<<n):  
		        current_lcm = 1  
		        for j in range(n):  
		            if (i >> j) & 1:  
		                current_lcm = lcm(current_lcm, coins[j])  
		        if i.bit_count() % 2 == 1:  
		            cnt += m // current_lcm  
		        else:  
		            cnt -= m // current_lcm  
		    return cnt >= k  
		  
		left = k - 1  
		right = min(coins) * k  
		while left + 1 < right:  
		    mid = (left + right) // 2  
		    if check(mid):  
		        right = mid  
		    else:  
		        left = mid  
		return right
```

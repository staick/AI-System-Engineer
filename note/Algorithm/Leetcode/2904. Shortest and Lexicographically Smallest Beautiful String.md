开始想用位运算，但是复杂度过高，超出了时间限制
```python
```python
class Solution:
    def shortestBeautifulSubstring(self, s: str, k: int) -> str:
        n = len(s)
        s = int(s, 2)
        if s.bit_count() < k:
            return ""
        ans = (1 << k) - 1
        while ans <= s:
            if ans.bit_count() != k:
                ans += 1
                continue
            for i in range(0, n - k + 1):
                if (s >> i) & ans == ans:
                    return f"{ans:b}"
            ans += 1
        return ""

看解析使用双指针来实现滑动窗口
```python
class Solution:
    def shortestBeautifulSubstring(self, s: str, k: int) -> str:
        if s.count('1') < k:
            return ''
        cnt = left = 0
        ans = s
        for right,digit in enumerate(s):
            cnt += int(digit)
            while cnt > k or s[left] == '0':
                cnt -= int(s[left])
                left += 1

            if cnt == k:
                cur = s[left:right + 1]

                if len(cur) < len(ans) or (len(cur) == len(ans) and cur < ans):
                    ans = cur
        return ans

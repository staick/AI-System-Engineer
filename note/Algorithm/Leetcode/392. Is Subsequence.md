class Solution:  
    def isSubsequence(self, s: str, t: str) -> bool:  
        first = 0  
        second = 0  
  
        while True:  
            if first == len(s):  
                return True  
            if second == len(t):  
                return False  
            if s[first] == t[second]:  
                first += 1  
                second += 1  
            else:  
                second += 1

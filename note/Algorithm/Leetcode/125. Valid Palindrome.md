```python
class Solution:  
    def isPalindrome(self, s: str) -> bool:  
        start = 0  
        end = len(s) - 1  
        flag = False  
        s = s.lower()  
        while start < end:  
            if not s[start].isalnum():  
                start += 1  
                continue  
            if not s[end].isalnum():  
                end -= 1  
                continue  
  
            if s[start] == s[end]:  
                start += 1  
                end -= 1  
            else:  
                break  
        else:  
            flag = True  
  
        return flag
```

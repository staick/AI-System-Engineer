```python
class Solution:
    def checkDivisibility(self, n: int) -> bool:
        s = 0
        c =  1
        n_copy = n
        while n:
            s, c = s + n % 10, c * (n % 10)
            n //= 10

        return n_copy % (s+c) == 0  
```
